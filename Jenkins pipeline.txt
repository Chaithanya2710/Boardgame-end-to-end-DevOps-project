pipeline {
    agent any
    
    tools{
        jdk 'JAVA_HOME'
        maven 'MAVEN_HOME'
    }
    
    environment {
        AWS_REGION = "ap-south-1"
        CLUSTER_NAME = "my-eks-cluser"
        DOCKER_IMAGE = "chaithanya1004/boardgame-image:v1"
    }

    stages {
        stage('Cloning the Boardgame project repository') {
            steps {
                git branch: 'main', url: 'https://github.com/Chaithanya2710/Boardgame.git'
            }
        }
        
        stage('Compiling the Boardgame project') {
            steps {
                sh 'mvn compile'
            }
        }
        
        stage('Test souce code of the Boardgame project') {
            steps {
                sh 'mvn test'
            }
        }
        
        stage('Scanning souce code of Boardgame project using Sonarqube') {
            steps {
                withSonarQubeEnv('Sonarqube') {
                    sh 'mvn verify org.sonarsource.scanner.maven:sonar-maven-plugin:sonar -Dsonar.projectName=boardgame -Dsonar.projectKey=boardgame'
                }
            }
        }
        
        stage('Deploying package to nexus') {
            steps {
                withMaven(globalMavenSettingsConfig: 'nexus-credentials', jdk: 'JAVA_HOME', maven: 'MAVEN_HOME', traceability: true) {
                    sh 'mvn clean deploy'
                }
            }
        }
        
        stage('Generating image for boardgame project using Docker') {
            steps {
                script {
                    withDockerRegistry(credentialsId: 'docker-credentials',url: 'https://index.docker.io/v1/') {
                        sh 'docker build -t chaithanya1004/boardgame-image:v1 .'
                    }
                }
            }
        }
        
        stage('Scanning image via Trivy') {
            steps {
                sh 'TMPDIR=/tmp-trivy trivy image --format table chaithanya1004/boardgame-image:v1'
            }
        }
        
        stage('Pushing the image to the dockerhub') {
            steps {
                script {
                    withDockerRegistry(credentialsId: 'docker-credentials',url: 'https://index.docker.io/v1/') {
                        sh 'docker push chaithanya1004/boardgame-image:v1'
                    }
                }
            }
        }
        
        stage('Deploying application to EKS cluster') {
            steps {
                withAWS(credentials: 'aws-credentials', region: 'ap-south-1') {
                    sh '''aws eks --region ${AWS_REGION} update-kubeconfig --name ${CLUSTER_NAME}
                          kubectl apply -f deployment-service.yaml'''
                }
            }
        }
    }
}
